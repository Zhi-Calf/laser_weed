cmake_minimum_required(VERSION 3.16)
project(row_nav_sdk VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(RK_NAV_BUILD_TESTS "Build SDK geometry and C ABI tests" ON)
option(RK_NAV_WITH_RKNN "Build RKNN inference backend" OFF)
option(ROW_NAV_BUILD_TOOLS "Build optional offline BEV calibration tools" OFF)
option(ROW_NAV_BUILD_LEGACY_CALIBRATION_TOOLS "Build legacy desktop calibration tools (aruco/highgui)" OFF)
set(ROW_NAV_SDK_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
list(APPEND CMAKE_MODULE_PATH "${ROW_NAV_SDK_SOURCE_DIR}/cmake")
set(ROW_NAV_OPENCV_COMPONENTS core calib3d dnn imgproc)
if(ROW_NAV_BUILD_TOOLS OR ROW_NAV_BUILD_LEGACY_CALIBRATION_TOOLS)
  list(APPEND ROW_NAV_OPENCV_COMPONENTS imgcodecs)
endif()
if(ROW_NAV_BUILD_LEGACY_CALIBRATION_TOOLS)
  list(APPEND ROW_NAV_OPENCV_COMPONENTS aruco highgui)
endif()
find_package(OpenCV 4 REQUIRED COMPONENTS ${ROW_NAV_OPENCV_COMPONENTS})
find_package(Threads REQUIRED)
include(cmake/RuntimeSources.cmake)
add_library(rk_nav_core STATIC ${ROW_NAV_RUNTIME_SOURCES})
set_target_properties(rk_nav_core PROPERTIES POSITION_INDEPENDENT_CODE ON
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_include_directories(rk_nav_core PUBLIC "${ROW_NAV_SDK_SOURCE_DIR}/include")
target_include_directories(rk_nav_core SYSTEM PUBLIC ${OpenCV_INCLUDE_DIRS})
target_link_libraries(rk_nav_core PUBLIC ${OpenCV_LIBS} Threads::Threads)
target_compile_options(rk_nav_core PRIVATE
  $<$<CXX_COMPILER_ID:GNU,Clang,AppleClang>:-Wall;-Wextra;-Wpedantic;-Wconversion>)
if(RK_NAV_WITH_RKNN)
  find_package(RKNN REQUIRED)
  target_sources(rk_nav_core PRIVATE src/inference/rknn_yolov8_segmenter.cpp)
  target_link_libraries(rk_nav_core PUBLIC RKNN::Runtime)
  target_compile_definitions(rk_nav_core PUBLIC RK_NAV_WITH_RKNN=1)
endif()
include(cmake/RowNavSDK.cmake)
if(ROW_NAV_BUILD_TOOLS)
  foreach(tool bev_calibrate bev_calibration_visualize)
    add_executable(${tool} apps/${tool}.cpp)
    target_link_libraries(${tool} PRIVATE rk_nav_core)
    install(TARGETS ${tool} RUNTIME DESTINATION bin)
  endforeach()
endif()
if(RK_NAV_BUILD_TESTS)
  include(CTest) # Generate DartConfiguration.tcl for colcon's CMake test runner.
  enable_testing()
  add_executable(rk_nav_unit_tests tests/test_main.cpp
    tests/unit/bev_calibration_test.cpp tests/unit/bev_projector_test.cpp
    tests/unit/camera_calibration_test.cpp tests/unit/centerline_fitter_test.cpp
    tests/unit/dynamic_bev_test.cpp tests/unit/multi_row_test.cpp
    tests/unit/perception_pipeline_test.cpp tests/unit/row_extractor_test.cpp
    tests/unit/row_temporal_tracker_test.cpp tests/unit/status_test.cpp)
  target_include_directories(rk_nav_unit_tests PRIVATE tests)
  target_link_libraries(rk_nav_unit_tests PRIVATE rk_nav_core)
  add_test(NAME rk_nav_unit_tests COMMAND rk_nav_unit_tests)
endif()

if(ROW_NAV_BUILD_LEGACY_CALIBRATION_TOOLS)
  foreach(tool charuco_calibrate camera_calibration_validate bev_point_picker)
    add_executable(${tool} apps/${tool}.cpp)
    target_link_libraries(${tool} PRIVATE rk_nav_core)
    install(TARGETS ${tool} RUNTIME DESTINATION bin)
  endforeach()
endif()
